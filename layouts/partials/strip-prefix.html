{{- /* 去掉导语固定开头（清单在 hugo.toml 的 [params] summary_strip_prefixes）。参数 dict "text" 文字 "site" 站点。 不分大小写；末尾有冒号的项全角或半角冒号都认，冒号及其后空白一并去掉；没有冒号的项后须接空白或正好在文末；只去第一个匹配的开头。 返回 dict：rest 去掉开头后的文字（未去时为原文）、stripped 是否去掉了开头、body 开头本身（不含冒号）、colon 该开头是否带冒号。 */ -}} {{- $text := .text -}} {{- $res := dict "rest" $text "stripped" false "body" "" "colon" false -}} {{- range .site.Params.summary_strip_prefixes -}} {{- if not (index $res "stripped") -}} {{- $hasColon := findRE `[：:]\s*$` . 1 -}} {{- $body := strings.TrimSpace (replaceRE `[：:]\s*$` "" .) -}} {{- $n := strings.RuneCount $body -}} {{- if and $body (ge (strings.RuneCount $text) $n) (eq (lower (substr $text 0 $n)) (lower $body)) -}} {{- $rest := substr $text $n -}} {{- $ok := false -}} {{- if $hasColon -}} {{- if findRE `^\s*[：:]` $rest 1 -}}{{- $rest = replaceRE `^\s*[：:]\s*` "" $rest 1 -}}{{- $ok = true -}}{{- end -}} {{- else if or (not $rest) (findRE `^\s` $rest 1) -}} {{- $rest = strings.TrimLeft " \t\n" $rest -}}{{- $ok = true -}} {{- end -}} {{- if $ok -}}{{- $res = dict "rest" $rest "stripped" true "body" $body "colon" (not (not $hasColon)) -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $res -}}
