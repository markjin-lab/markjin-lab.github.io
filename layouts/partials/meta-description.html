{{- /* 页面描述（），中英文各取本语言的 i18n 和正文。取值优先级： 1. front matter 里的 description；2. i18n 的 meta_desc_<键>：首页 home，404 页 404，其余按路径（/posts/daily → posts_daily，标签页 tags、分类页 categories，依此类推）； 3. 普通文章（Kind 为 page）没有对应键时取正文第一段（日报即导语），这一段的第一句不超长就只用第一句；4. 仍为空时用首页描述（meta_desc_home）。 以下处理对所有描述生效：去掉出处编号（[1]、链接文字为数字的链接，及相邻的一串编号）、Markdown 链接只留文字、去 HTML 标签并还原实体、合并空白； 超过上限（中文 120 字、英文 200 字符）才截短：按上限截断并加「…」，中文退到最后一个逗号、分号、冒号或顿号处， 英文去掉未闭合的括号和结尾最多三个介词、连词、冠词等短词。 */ -}} {{- $desc := "" -}} {{- $fromContent := false -}} {{- with .Description -}} {{- $desc = . -}} {{- else -}} {{- $key := "home" -}} {{- if eq .Kind "404" -}} {{- $key = "404" -}} {{- else if not .IsHome -}} {{- $key = replace (strings.Trim .Path "/") "/" "_" -}} {{- end -}} {{- $desc = i18n (printf "meta_desc_%s" $key) -}} {{- if and (not $desc) (eq .Kind "page") -}} {{- with findRE `(?s)

.*?

` .Content 1 -}} {{- $desc = index . 0 -}} {{- $fromContent = true -}} {{- end -}} {{- end -}} {{- if not $desc -}}{{- $desc = i18n "meta_desc_home" -}}{{- end -}} {{- end -}} {{- $desc = replaceRE `\[]*>(\d+)\]` "[$1]" $desc -}} {{- $desc = replaceRE `]*>\[?(\d+)\]?` "[$1]" $desc -}} {{- $desc = replaceRE `\s*\[\d+\](\s*[、,，]?\s*\[\d+\])*` "" $desc -}} {{- $desc = replaceRE `\[([^\]]+)\]\([^)]+\)` "$1" $desc -}} {{- $desc = $desc | plainify | htmlUnescape | strings.TrimSpace -}} {{- $desc = replaceRE `\s+` " " $desc -}} {{- $limit := cond (eq .Language.Lang "zh") 120 200 -}} {{- if $fromContent -}} {{- $sentenceRE := cond (eq .Language.Lang "zh") `^.*?[。！？]` `^.*?[.!?](?:\s|$)` -}} {{- with findRE $sentenceRE $desc 1 -}} {{- $first := strings.TrimSpace (index . 0) -}} {{- if le (strings.RuneCount $first) $limit -}} {{- $desc = $first -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt (strings.RuneCount $desc) $limit -}} {{- $desc = truncate (sub $limit 1) "…" $desc | htmlUnescape -}} {{- if eq .Language.Lang "zh" -}} {{- $desc = replaceRE `[，,；;：:、][^，,；;：:、。！？]*…$` "…" $desc -}} {{- else -}} {{- $desc = replaceRE `\([^()]*…$` "…" $desc -}} {{- $desc = replaceRE `(?i)(?:\s+(?:a|an|and|as|at|by|even|for|from|in|into|of|on|or|the|to|when|with)){1,3}…$` "…" $desc -}} {{- $desc = replaceRE `\s+…$` "…" $desc -}} {{- end -}} {{- end -}} {{- with $desc }}{{ end -}}
